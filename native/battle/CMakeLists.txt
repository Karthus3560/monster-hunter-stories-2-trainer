cmake_minimum_required(VERSION 3.20)
project(KinshipNativeBattleV2 LANGUAGES CXX)
add_library(KinshipNativeBattleV2 SHARED native_battle.cpp)
target_compile_features(KinshipNativeBattleV2 PRIVATE cxx_std_20)
target_compile_definitions(KinshipNativeBattleV2 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(KinshipNativeBattleV2 PRIVATE /W4 /EHsc /guard:cf)
target_link_libraries(KinshipNativeBattleV2 PRIVATE bcrypt windowscodecs ole32)
set_property(TARGET KinshipNativeBattleV2 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(BattleEngineChecks engine_checks.cpp)
target_compile_features(BattleEngineChecks PRIVATE cxx_std_20)
target_compile_definitions(BattleEngineChecks PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BattleEngineChecks PRIVATE /W4 /EHsc)
set_property(TARGET BattleEngineChecks PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(NativeStudioChecks studio_checks.cpp)
target_compile_features(NativeStudioChecks PRIVATE cxx_std_20)
target_compile_definitions(NativeStudioChecks PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(NativeStudioChecks PRIVATE /W4 /EHsc)
target_link_libraries(NativeStudioChecks PRIVATE bcrypt)
