cmake_minimum_required(VERSION 3.20)
project(KinshipCampIcon LANGUAGES CXX)
add_library(KinshipCampIcon SHARED icon.cpp)
target_compile_features(KinshipCampIcon PRIVATE cxx_std_20)
target_compile_definitions(KinshipCampIcon PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(KinshipCampIcon PRIVATE /W4 /EHsc /guard:cf)
target_link_libraries(KinshipCampIcon PRIVATE bcrypt windowscodecs ole32)
set_property(TARGET KinshipCampIcon PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_custom_command(TARGET KinshipCampIcon POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../assets/kinship-emblem.png" "$<TARGET_FILE_DIR:KinshipCampIcon>/kinship-emblem.png")
